446 // Interal function that does most of the work for aviationMapsChanged()
447 // emits geoJSONChanged() when done. This function is meant to be run in a
448 // separate thread.
449 // Lookup structures over _waypoints_, built together with the waypoint
450 // list on the worker thread and replaced together with it. Rows are
451 // indices into _waypoints_.
452 struct WaypointIndex {
453 // First row (in name order) for every non-empty ICAO code
454 QHash<QString, int> rowByICAOCode;
455 // Per row: simplify(name) + ' ' + simplify(ICAO code)
456 QStringList searchKeys;
457 // (cell, row) pairs sorted by cell; see cellOf()
458 QList<std::pair<qint32, int>> cells;
459 };

461 QList<Waypoint> waypoints;
462 WaypointIndex index;
463 QList<Airspace> airspaces;
464 QByteArray combinedGeoJSON;
465 };
